เพิ่มไว้ในไฟล์เซิร์ฟเวอร์ ก่อน Route ทั้งหมด

function validateEvaluatee(body) {
  const errors = [];
  if (!body.name || body.name.trim() === '') {
    errors.push('กรุณากรอกชื่อ-สกุล');
  } else if (body.name.length > 100) {
    errors.push('ชื่อ-สกุลยาวเกิน 100 ตัวอักษร');
  }
  if (body.department && body.department.length > 100) {
    errors.push('ชื่อแผนกยาวเกิน 100 ตัวอักษร');
  }
  return errors;
}

ใน Route POST /evaluatees เดิม เพิ่มสามบรรทัดนี้เป็นบรรทัดแรกในฟังก์ชัน ก่อน try

  const errors = validateEvaluatee(req.body);
  if (errors.length > 0) {
    return res.status(400).json({ message: errors.join(' / ') });
  }

ผ่านเมื่อ POST ด้วย {"name": ""} ได้ 400 พร้อมข้อความ กรุณากรอกชื่อ-สกุล

PUT แก้ไข

app.put('/evaluatees/:id', async (req, res) => {
  const errors = validateEvaluatee(req.body);
  if (errors.length > 0) {
    return res.status(400).json({ message: errors.join(' / ') });
  }
  try {
    const { name, department } = req.body;
    const [result] = await pool.query(
      'UPDATE evaluatees SET name = ?, department = ? WHERE id = ?',
      [name, department, req.params.id]
    );
    if (result.affectedRows === 0) {
      return res.status(404).json({ message: 'ไม่พบผู้รับการประเมิน' });
    }
    res.json({ id: Number(req.params.id), name, department });
  } catch (err) {
    console.error(err);
    res.status(500).json({ message: 'เกิดข้อผิดพลาดที่ฐานข้อมูล' });
  }
});

ผ่านเมื่อ PUT /evaluatees/1 แล้ว GET ได้ค่าใหม่, PUT id 999 ได้ 404, PUT ชื่อว่างได้ 400

async function removeEvaluatee(id) {
  if (!confirm('ยืนยันการลบ?')) return

หน้าเว็บ: ใช้ฟอร์มเดิมแก้ไขได้

แทนที่ฟังก์ชัน addEvaluatee เดิมด้วยชุดนี้ ฟอร์มเดียวทำได้ทั้งเพิ่มและแก้ไข

const editingId = ref(null)

function startEdit(e) {
  editingId.value = e.id
  form.value = { name: e.name, department: e.department }
}

function cancelEdit() {
  editingId.value = null
  form.value = { name: '', department: '' }
}

async function saveEvaluatee() {
  errorMsg.value = ''
  const url = editingId.value
    ? `${API}/evaluatees/${editingId.value}`
    : `${API}/evaluatees`
  const method = editingId.value ? 'PUT' : 'POST'
  try {
    const res = await fetch(url, {
      method,
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify(form.value)
    })
    const data = await res.json()
    if (!res.ok) throw new Error(data.message)
    cancelEdit()
    await loadEvaluatees()
  } catch (err) {
    errorMsg.value = err.message
  }
}

แก้ฟอร์มให้เรียก saveEvaluatee และเปลี่ยนปุ่มตามโหมด

<form @submit.prevent="saveEvaluatee">
  <input v-model="form.name" placeholder="ชื่อ-สกุล">
  <input v-model="form.department" placeholder="แผนก">
  <button type="submit">{{ editingId ? 'บันทึกการแก้ไข' : 'เพิ่ม' }}</button>
  <button v-if="editingId" type="button" @click="cancelEdit">ยกเลิก</button>
</form>

เราเอา required ออกจากช่องชื่อโดยตั้งใจ จะได้เห็นว่าข้อความ 400 จาก Express ขึ้นบนหน้าเว็บจริง

ผ่านเมื่อ กดแก้ไขแล้วข้อมูลขึ้นในฟอร์ม กดบันทึกแล้วตารางเปลี่ยน, ล้างชื่อแล้วกดบันทึกเห็นข้อความ กรุณากรอกชื่อ-สกุล, กดลบแล้วมีถามยืนยัน จากนั้น commit และ push
