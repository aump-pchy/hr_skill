ส่วนที่ 2 (60 นาที)

ออกแบบฐานข้อมูล

เรื่องใหม่: Foreign Key

Foreign Key คือคอลัมน์ที่เก็บ id ของอีกตาราง เพื่อบอกว่าแถวนี้เป็นของใคร เช่น ตัวชี้วัดมี topic_id บอกว่าอยู่ใต้หัวข้อไหน MySQL จะไม่ยอมให้ใส่ topic_id ที่ไม่มีอยู่จริง ข้อมูลจึงไม่หลุดลอย

FOREIGN KEY (topic_id) REFERENCES topics(id) ON DELETE CASCADE

ON DELETE CASCADE คือลบหัวข้อแล้วตัวชี้วัดใต้หัวข้อนั้นหายตามไปด้วย ใช้เมื่อข้อมูลลูกไม่มีความหมายถ้าไม่มีแม่

1

หาตาราง

ไล่จาก Use Case ว่ามีข้อมูลอะไรต้องเก็บ คำนามที่โผล่บ่อยมักเป็นตาราง เช่น รอบประเมิน หัวข้อ ตัวชี้วัด ผลงาน หลักฐาน ตั้งชื่อตารางเป็นภาษาอังกฤษตัวเล็ก พหูพจน์ คั่นคำด้วยขีดล่าง

2

ใส่คอลัมน์และชนิดข้อมูล

ทุกตารางมี id เป็น PRIMARY KEY เลือกชนิดข้อมูลให้เหมาะ คะแนนแบบมี/ไม่มี และแบบ 1–4 เก็บต่างกันอย่างไร ทีมต้องตกลงกัน

3

ลากเส้นความสัมพันธ์

ใช้ ER Diagram ใน Draw.io บันทึกเป็น docs/er.drawio ทุกเส้นต้องบอกได้ว่าเป็นหนึ่งต่อหลายหรือหลายต่อหลาย

4

เขียนเป็น schema.sql

คนแกนเขียนไฟล์ backend/sql/schema.sql ที่สร้างฐานข้อมูลใหม่หมดได้ในคำสั่งเดียว ขึ้นต้นด้วยสามบรรทัดนี้ แล้วตามด้วย CREATE TABLE ทุกตาราง เรียงตารางแม่ก่อนตารางลูก

DROP DATABASE IF EXISTS evaluation;
CREATE DATABASE evaluation CHARACTER SET utf8mb4;
USE evaluation;

รันไฟล์นี้จากใน MySQL ได้ โดยเปิด terminal ที่โฟลเดอร์ backend ก่อนเข้า MySQL

SOURCE sql/schema.sql;

ผ่านเมื่อ รัน SOURCE ได้โดยไม่มี error และ SHOW TABLES; เห็นครบทุกตารางตามแผนภาพ

แบบอ้างอิงของจารย์ เปิดหลังทีมออกแบบเองเสร็จแล้วเท่านั้น

ใช้เทียบว่าทีมขาดอะไรหรือมีอะไรที่ทำต่างไป ทีมไม่จำเป็นต้องเหมือนแบบนี้ แต่ต้องอธิบายได้ว่าทำไมออกแบบแบบของตัวเอง

DROP DATABASE IF EXISTS evaluation;
CREATE DATABASE evaluation CHARACTER SET utf8mb4;
USE evaluation;

CREATE TABLE users (
  id INT AUTO_INCREMENT PRIMARY KEY,
  username VARCHAR(50) NOT NULL UNIQUE,
  password_hash VARCHAR(255) NOT NULL,
  full_name VARCHAR(100) NOT NULL,
  department VARCHAR(100),
  role ENUM('hr','evaluatee','committee') NOT NULL,
  must_change_password BOOLEAN NOT NULL DEFAULT TRUE,
  created_at DATETIME DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE evaluation_rounds (
  id INT AUTO_INCREMENT PRIMARY KEY,
  name VARCHAR(100) NOT NULL,
  start_date DATE NOT NULL,
  end_date DATE NOT NULL
);

CREATE TABLE topics (
  id INT AUTO_INCREMENT PRIMARY KEY,
  round_id INT NOT NULL,
  title VARCHAR(200) NOT NULL,
  description TEXT,
  FOREIGN KEY (round_id) REFERENCES evaluation_rounds(id) ON DELETE CASCADE
);

CREATE TABLE indicators (
  id INT AUTO_INCREMENT PRIMARY KEY,
  topic_id INT NOT NULL,
  title VARCHAR(200) NOT NULL,
  description TEXT,
  weight INT NOT NULL,
  score_type ENUM('scale_1_4','yes_no') NOT NULL,
  FOREIGN KEY (topic_id) REFERENCES topics(id) ON DELETE CASCADE
);

CREATE TABLE assignments (
  id INT AUTO_INCREMENT PRIMARY KEY,
  round_id INT NOT NULL,
  evaluatee_id INT NOT NULL,
  committee_id INT NOT NULL,
  committee_role ENUM('chair','member') NOT NULL,
  status ENUM('draft','submitted') NOT NULL DEFAULT 'draft',
  overall_comment TEXT,
  signature_path VARCHAR(255),
  submitted_at DATETIME,
  UNIQUE (round_id, evaluatee_id, committee_id),
  FOREIGN KEY (round_id) REFERENCES evaluation_rounds(id) ON DELETE CASCADE,
  FOREIGN KEY (evaluatee_id) REFERENCES users(id),
  FOREIGN KEY (committee_id) REFERENCES users(id)
);

CREATE TABLE works (
  id INT AUTO_INCREMENT PRIMARY KEY,
  evaluatee_id INT NOT NULL,
  indicator_id INT NOT NULL,
  title VARCHAR(200) NOT NULL,
  detail TEXT,
  created_at DATETIME DEFAULT CURRENT_TIMESTAMP,
  FOREIGN KEY (evaluatee_id) REFERENCES users(id),
  FOREIGN KEY (indicator_id) REFERENCES indicators(id) ON DELETE CASCADE
);

CREATE TABLE evidences (
  id INT AUTO_INCREMENT PRIMARY KEY,
  work_id INT NOT NULL,
  type ENUM('file','url') NOT NULL,
  file_path VARCHAR(255),
  original_name VARCHAR(255),
  url VARCHAR(500),
  FOREIGN KEY (work_id) REFERENCES works(id) ON DELETE CASCADE
);

CREATE TABLE self_assessments (
  id INT AUTO_INCREMENT PRIMARY KEY,
  evaluatee_id INT NOT NULL,
  indicator_id INT NOT NULL,
  score INT NOT NULL,
  note TEXT,
  UNIQUE (evaluatee_id, indicator_id),
  FOREIGN KEY (evaluatee_id) REFERENCES users(id),
  FOREIGN KEY (indicator_id) REFERENCES indicators(id) ON DELETE CASCADE
);

CREATE TABLE committee_scores (
  id INT AUTO_INCREMENT PRIMARY KEY,
  assignment_id INT NOT NULL,
  indicator_id INT NOT NULL,
  score INT NOT NULL,
  comment TEXT,
  UNIQUE (assignment_id, indicator_id),
  FOREIGN KEY (assignment_id) REFERENCES assignments(id) ON DELETE CASCADE,
  FOREIGN KEY (indicator_id) REFERENCES indicators(id) ON DELETE CASCADE
);

คะแนนแบบมี/ไม่มีเก็บเป็น 1 กับ 0 ในคอลัมน์ score เดียวกับแบบ 1–4 แล้วดูชนิดจาก score_type ของตัวชี้วัด ส่วนสถานะร่างกับยืนยันและลายเซ็นของกรรมการอยู่ใน assignments เพราะกรรมการหนึ่งคนยืนยันผลของผู้รับการประเมินหนึ่งคนครั้งเดียว

1

ตกลงกติกากลางของทีม

เขียนไว้บนสุดของ docs/api.md ให้ทุกคนทำเหมือนกัน

## กติกากลาง
- ชื่อ field ใน JSON ใช้ชื่อเดียวกับคอลัมน์ในฐานข้อมูล เช่น full_name, topic_id
- error ทุกตัวตอบเป็น { "message": "ข้อความภาษาไทย" }
- ทุก Endpoint ยกเว้น /auth/login ต้องแนบ Authorization: Bearer <token>
- Path ขึ้นต้นตามบทบาท: /hr/... งานบุคลากร, /my/... ผู้รับการประเมิน, /committee/... กรรมการ
- /account/... ใช้ได้ทุกบทบาทที่ Login แล้ว
2

เขียนตาราง Endpoint

ใช้รูปแบบเดียวกันทุกแถว เกณฑ์ให้คะแนนต้องการ Endpoint, Method, Parameter, Response, Status Code, Error และสิทธิ์

## งานบุคลากร
| Method | Endpoint | สิทธิ์ | ส่งไป | สำเร็จได้ | Error |
|---|---|---|---|---|---|
| POST | /auth/login | ไม่ต้อง Login | username, password | 200 token, user | 400, 401 |
| GET | /hr/rounds | hr | - | 200 รายการรอบ | 401, 403 |
| POST | /hr/rounds | hr | name, start_date, end_date | 201 รอบที่สร้าง | 400, 401, 403 |

ผ่านเมื่อ ทุกฟังก์ชันใน Use Case ของกลุ่มตัวเองมีอย่างน้อยหนึ่ง Endpoint รองรับ และมีครบ GET, POST, PUT, DELETE ในภาพรวมของทั้งระบบ

แยก Route เป็นไฟล์ด้วย express.Router

แต่ละไฟล์ใน routes/ หน้าตาแบบนี้ ใส่ด่านตรวจไว้บนสุดครั้งเดียว ทุก Route ในไฟล์ได้ด่านเดียวกัน

const express = require('express');
const pool = require('../db');
const { requireAuth, requireRole } = require('../middleware/auth');

const router = express.Router();
router.use(requireAuth, requireRole('hr'));

router.get('/rounds', async (req, res) => {
  // ...
});

module.exports = router;

แล้วใน server.js เหลือแค่การตั้งค่าและการบอกว่า path ไหนใช้ไฟล์ไหน

app.use('/auth', require('./routes/auth'));
app.use('/hr', require('./routes/hr'));
app.use('/account', require('./routes/account'));
app.use('/my', require('./routes/evaluatee'));
app.use('/committee', require('./routes/committee'));

ใน routes/auth.js path จะเหลือแค่ /login เพราะ /auth ถูกใส่ให้จาก server.js แล้ว ไฟล์ที่ยังไม่มีงาน ให้สร้างไว้ก่อนโดยมีแค่ router เปล่ากับ module.exports

กติกา Git ของทีม และฝึกแก้ conflict

ทุกคนทำงานบน main ทำตามลำดับนี้ทุกครั้ง

git pull            ก่อนเริ่มทำงานทุกครั้ง
(ทำงาน ทดสอบให้ผ่าน)
git add .
git commit -m "บอกว่าทำอะไร"
git pull            ดึงของเพื่อนมาก่อนส่ง
git push

ไฟล์ที่ใช้ร่วมกัน เช่น server.js, router.js และ schema.sql ให้คนแกนแก้คนเดียว ใครต้องการเพิ่มอะไร บอกคนแกน และหลังคนแกน push ไฟล์เหล่านี้ ทุกคน git pull ทันที

ฝึก conflict หนึ่งรอบ ทุกคนแก้บรรทัดแรกของ README.md เป็นชื่อตัวเอง แล้ว commit และ push ตามกันทีละคน คนที่สองเป็นต้นไปจะเจอ conflict ตอน git pull ให้เปิดไฟล์ใน VS Code กด Accept Both Changes แล้ว commit และ push

ผ่านเมื่อ ทุกคนแก้ conflict เองได้หนึ่งครั้ง และ README.md บน GitLab มีชื่อครบทุกคน
