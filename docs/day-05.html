ห้ามเก็บรหัสผ่านตรง ๆ

ถ้าฐานข้อมูลหลุด คนร้ายจะเห็นรหัสทุกคนทันที เราจึงเก็บแค่ hash ของรหัสผ่าน คือผลจากการแปลงรหัสแบบทางเดียว แปลงกลับไม่ได้ ตอนเข้าสู่ระบบ bcrypt จะเอารหัสที่พิมพ์มาเทียบกับ hash ให้ว่าตรงกันไหม

token คือบัตรผ่าน

ส่งชื่อผู้ใช้และรหัสผ่านถูกต้องExpress ออก tokenหน้าเว็บเก็บไว้แนบ token ทุกครั้งที่เรียก API

JWT เป็น token แบบหนึ่ง ข้างในมีข้อมูลเช่น id และบทบาทของผู้ใช้ และมีลายเซ็นที่สร้างจากรหัสลับของเซิร์ฟเวอร์ ใครแก้ข้อมูลข้างใน ลายเซ็นจะไม่ตรงทันที แต่ข้อมูลข้างในใครก็อ่านได้ จึงห้ามใส่รหัสผ่านหรือข้อมูลลับลงไป

ชนิดข้อมูลใหม่

ENUM('hr','evaluatee','committee') คือคอลัมน์ที่รับได้แค่ค่าในรายการ ใช้กับบทบาทของผู้ใช้ ส่วน BOOLEAN เก็บจริงหรือเท็จ ใช้จำว่าต้องเปลี่ยนรหัสครั้งแรกหรือยัง และ UNIQUE คือห้ามซ้ำ ใช้กับชื่อผู้ใช้

Status code ใหม่

Codeใช้เมื่อ
401ยืนยันตัวตนไม่ผ่าน เช่น รหัสผ่านผิด หรือไม่มี token
409ข้อมูลชนกับที่มีอยู่แล้ว เช่น ชื่อผู้ใช้ซ้ำ

สมัครผู้ใช้

ในระบบจริง งานบุคลากรเป็นคนเพิ่มผู้ใช้ วันนี้ทำ Route นี้ไว้ก่อนเพื่อสร้างผู้ใช้ทดสอบ

app.post('/auth/register', async (req, res) => {
  const { username, password, full_name, role } = req.body;
  if (!username || !password || !full_name || !role) {
    return res.status(400).json({ message: 'กรุณากรอกข้อมูลให้ครบ' });
  }
  if (password.length < 8) {
    return res.status(400).json({ message: 'รหัสผ่านต้องยาวอย่างน้อย 8 ตัวอักษร' });
  }
  if (!['hr', 'evaluatee', 'committee'].includes(role)) {
    return res.status(400).json({ message: 'บทบาทไม่ถูกต้อง' });
  }
  try {
    const passwordHash = await bcrypt.hash(password, 10);
    const [result] = await pool.query(
      'INSERT INTO users (username, password_hash, full_name, role) VALUES (?, ?, ?, ?)',
      [username, passwordHash, full_name, role]
    );
    res.status(201).json({ id: result.insertId, username, full_name, role });
  } catch (err) {
    if (err.code === 'ER_DUP_ENTRY') {
      return res.status(409).json({ message: 'ชื่อผู้ใช้นี้มีแล้ว' });
    }
    console.error(err);
    res.status(500).json({ message: 'เกิดข้อผิดพลาดที่ฐานข้อมูล' });
  }
});

สมัครผู้ใช้ทดสอบ 3 คนด้วย Thunder Client รหัสผ่าน password123 ทุกคน ใช้ชื่อผู้ใช้ hr1, eva1 และ com1 ตามบทบาท

{
  "username": "hr1",
  "password": "password123",
  "full_name": "ฝ่ายบุคลากร ทดสอบ",
  "role": "hr"
}

ผ่านเมื่อ สมัครได้ 201 ครบ 3 คน, สมัคร hr1 ซ้ำได้ 409, และ SELECT username, password_hash FROM users; เห็น hash ที่ขึ้นต้นด้วย $2 ไม่ใช่รหัสผ่านจริง จากนั้น commit และ push

1

Route เข้าสู่ระบบ

app.post('/auth/login', async (req, res) => {
  const { username, password } = req.body;
  if (!username || !password) {
    return res.status(400).json({ message: 'กรุณากรอกชื่อผู้ใช้และรหัสผ่าน' });
  }
  try {
    const [rows] = await pool.query(
      'SELECT * FROM users WHERE username = ?',
      [username]
    );
    const user = rows[0];
    const passwordOk = user && await bcrypt.compare(password, user.password_hash);
    if (!passwordOk) {
      return res.status(401).json({ message: 'ชื่อผู้ใช้หรือรหัสผ่านไม่ถูกต้อง' });
    }
    const token = jwt.sign(
      { id: user.id, username: user.username, role: user.role },
      JWT_SECRET,
      { expiresIn: '8h' }
    );
    res.json({
      token,
      user: {
        id: user.id,
        username: user.username,
        full_name: user.full_name,
        role: user.role,
        must_change_password: Boolean(user.must_change_password)
      }
    });
  } catch (err) {
    console.error(err);
    res.status(500).json({ message: 'เกิดข้อผิดพลาดที่ฐานข้อมูล' });
  }
});

สังเกตว่าชื่อผู้ใช้ไม่มีกับรหัสผ่านผิด ได้ข้อความเดียวกัน เพื่อไม่บอกคนร้ายว่าชื่อผู้ใช้ไหนมีอยู่จริง

ผ่านเมื่อ Login ด้วย hr1 ถูกต้องได้ token และข้อมูลผู้ใช้, รหัสผิดได้ 401, ชื่อผู้ใช้ที่ไม่มีได้ 401 ข้อความเดียวกัน

หน้าเข้าสู่ระบบด้วย Vue

ทำใน frontend/src/App.vue หน้าเดิม ถ้ายังไม่ได้เข้าสู่ระบบ ให้เห็นแค่ฟอร์ม Login ถ้าเข้าแล้ว ค่อยเห็นของเดิมทั้งหมด

1

ตัวแปรและฟังก์ชันเข้า ออกระบบ

เพิ่มใน <script setup>

const user = ref(JSON.parse(localStorage.getItem('user') || 'null'))
const loginForm = ref({ username: '', password: '' })
const loginError = ref('')

async function login() {
  loginError.value = ''
  try {
    const res = await fetch(`${API}/auth/login`, {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify(loginForm.value)
    })
    const data = await res.json()
    if (!res.ok) throw new Error(data.message)
    localStorage.setItem('token', data.token)
    localStorage.setItem('user', JSON.stringify(data.user))
    user.value = data.user
    loginForm.value = { username: '', password: '' }
    await loadEvaluatees()
  } catch (err) {
    loginError.value = err.message
  }
}

function logout() {
  localStorage.removeItem('token')
  localStorage.removeItem('user')
  user.value = null
}

แล้วแก้ onMounted เดิม ให้โหลดข้อมูลเฉพาะตอนเข้าสู่ระบบแล้ว

onMounted(() => {
  if (user.value) loadEvaluatees()
})
2

ครอบหน้าเดิมด้วยเงื่อนไข

ใน <template> ให้ <main> มีสองส่วน ส่วนแรกคือฟอร์ม Login ส่วนที่สองคือของเดิมทั้งหมด

<main>
  <section v-if="!user">
    <h1>เข้าสู่ระบบ</h1>
    <form @submit.prevent="login">
      <input v-model="loginForm.username" placeholder="ชื่อผู้ใช้" required>
      <input v-model="loginForm.password" type="password" placeholder="รหัสผ่าน" required>
      <button type="submit">เข้าสู่ระบบ</button>
    </form>
    <p v-if="loginError" style="color: red">{{ loginError }}</p>
  </section>

  <section v-else>
    <p>
      สวัสดี {{ user.full_name }} ({{ user.role }})
      <button @click="logout">ออกจากระบบ</button>
    </p>
    <!-- ของเดิมทั้งหมด ตั้งแต่ h1 รายชื่อผู้รับการประเมิน ลงมา -->
  </section>
</main>

ผ่านเมื่อ Login จากหน้าเว็บแล้วเห็นชื่อและบทบาท, รีเฟรชหน้ายังอยู่ในระบบ, กดออกแล้วกลับมาหน้า Login, ใส่รหัสผิดเห็นข้อความสีแดง จากนั้น commit และ push
