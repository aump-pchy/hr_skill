สูตรคะแนน (อยู่ใน utils/score.js)

  • yes_no: คะแนนถ่วง = score × weight (มี = 1, ไม่มี = 0)
  • scale_4: คะแนนถ่วง = score ÷ 4 × weight
  • ร้อยละรวม = ผลรวมคะแนนถ่วง ÷ ผลรวม weight × 100
  • ร้อยละการบันทึก = ตัวชี้วัดที่มีคะแนนตนเองแล้ว ÷ ตัวชี้วัดทั้งหมด × 100

import { ref } from 'vue'
import { useRouter } from 'vue-router'
import api, { errorMessage } from '../api'
import { getUser, updateUser, homePath } from '../auth'

const router = useRouter()
const form = ref({ old_password: '', new_password: '', confirm: '' })
const error = ref('')

async function save() {
  error.value = ''
  if (form.value.new_password.length < 6) { error.value = 'รหัสใหม่ต้องยาวอย่างน้อย 6 ตัวอักษร'; return }
  if (form.value.new_password !== form.value.confirm) { error.value = 'ยืนยันรหัสผ่านไม่ตรงกัน'; return }
  try {
    await api.put('/auth/change-password', {
      old_password: form.value.old_password,
      new_password: form.value.new_password,
    })
    updateUser({ must_change_password: 0 })
    router.push(homePath(getUser().role))
  } catch (err) {
    error.value = errorMessage(err)
  }
}

router.get('/indicators', async (req, res) => {
  try {
    const topic = await openTopic();
    if (!topic) return fail(res, 404, 'ยังไม่เปิดการประเมิน');

    const [indicators] = await db.query('SELECT * FROM indicators WHERE topic_id = ? ORDER BY id', [topic.id]);
    const [selfScores] = await db.query('SELECT indicator_id, score FROM self_scores WHERE evaluatee_id = ?', [req.user.id]);
    const [evidences] = await db.query('SELECT * FROM evidences WHERE evaluatee_id = ? ORDER BY id', [req.user.id]);

    const items = indicators.map((i) => ({
      ...i,
      self_score: selfScores.find((s) => s.indicator_id === i.id)?.score ?? null,
      evidences: evidences.filter((e) => e.indicator_id === i.id),
    }));
    ok(res, { topic, indicators: items });
  } catch (e) {
    fail(res, 500, e.message);
  }
});

router.put('/self-scores/:indicatorId', async (req, res) => {
  try {
    const [rows] = await db.query('SELECT * FROM indicators WHERE id = ?', [req.params.indicatorId]);
    if (!rows.length) return fail(res, 404, 'ไม่พบตัวชี้วัด');
    const score = Number(req.body.score);
    const valid = rows[0].eval_type === 'yes_no' ? [0, 1].includes(score) : score >= 1 && score <= 4;
    if (!valid) return fail(res, 400, 'คะแนนไม่อยู่ในช่วงที่กำหนด');

      `INSERT INTO self_scores (evaluatee_id, indicator_id, score) VALUES (?, ?, ?)
       ON DUPLICATE KEY UPDATE score = VALUES(score)`,
      [req.user.id, req.params.indicatorId, score]
    );
    ok(res, null, 'บันทึกคะแนนสำเร็จ');

function formOf(id) {

  return forms.value[id]

async function addEvidence(indicatorId) {
  const f = formOf(indicatorId)
  if (!f.detail.trim()) { error.value = 'กรุณากรอกรายละเอียด'; return }
  const fd = new FormData()
  fd.append('indicator_id', indicatorId)
  fd.append('detail', f.detail)
  if (f.url) fd.append('url', f.url)
  if (f.file) fd.append('file', f.file)
  try {
    await api.post('/evaluatee/evidences', fd)
    forms.value[indicatorId] = { detail: '', url: '', file: null }
    load()
  } catch (err) {
    error.value = errorMessage(err)
  }
}
