ฐานข้อมูลจริงและผู้ใช้ทดสอบ

ทุกคนรอสองไฟล์นี้อยู่ ทำให้เสร็จก่อนงานอื่นทั้งหมด สร้าง backend/seed.js ใส่ผู้ใช้ทดสอบ 3 บทบาท รอบประเมิน หัวข้อ ตัวชี้วัด และการมอบหมายหนึ่งคู่

const bcrypt = require('bcryptjs');
const pool = require('./db');

async function seed() {
  const hash = await bcrypt.hash('password123', 10);
  await pool.query(
    'INSERT INTO users (username, password_hash, full_name, department, role, must_change_password) VALUES ?',
    [[
      ['hr1', hash, 'ฝ่ายบุคลากร ทดสอบ', 'งานบุคลากร', 'hr', false],
      ['eva1', hash, 'สมชาย ใจดี', 'IT', 'evaluatee', true],
      ['com1', hash, 'วิภา ตั้งใจ', 'IT', 'committee', false]
    ]]
  );
  const [round] = await pool.query(
    'INSERT INTO evaluation_rounds (name, start_date, end_date) VALUES (?, ?, ?)',
    ['รอบที่ 1/2569', '2026-11-01', '2026-11-30']
  );
  const [topic] = await pool.query(
    'INSERT INTO topics (round_id, title) VALUES (?, ?)',
    [round.insertId, 'ด้านการจัดการเรียนการสอน']
  );
  await pool.query(
    'INSERT INTO indicators (topic_id, title, weight, score_type) VALUES ?',
    [[
      [topic.insertId, 'การวางแผนการสอน', 10, 'scale_1_4'],
      [topic.insertId, 'มีแผนการสอนครบทุกรายวิชา', 5, 'yes_no']
    ]]
  );
  const [users] = await pool.query(
    "SELECT id, username FROM users WHERE username IN ('eva1', 'com1')"
  );
  const idOf = name => users.find(u => u.username === name).id;
  await pool.query(
    'INSERT INTO assignments (round_id, evaluatee_id, committee_id, committee_role) VALUES (?, ?, ?, ?)',
    [round.insertId, idOf('eva1'), idOf('com1'), 'chair']
  );
  console.log('ใส่ข้อมูลทดสอบเรียบร้อย');
  process.exit(0);
}

seed().catch(err => {
  console.error(err);
  process.exit(1);
});

const [topics] = await pool.query('SELECT * FROM topics WHERE round_id = ?', [req.params.id]);
const [indicators] = await pool.query(
  'SELECT i.* FROM indicators i JOIN topics t ON i.topic_id = t.id WHERE t.round_id = ?',
  [req.params.id]
);
res.json(topics.map(t => ({
  ...t,
  indicators: indicators.filter(i => i.topic_id === t.id)
})));

คนผู้รับการประเมิน

เปลี่ยนรหัสครั้งแรก ข้อมูลส่วนตัว และผลงาน

Express สองไฟล์ routes/account.js ใช้ requireAuth อย่างเดียว ส่วน routes/evaluatee.js ใช้ได้เฉพาะ evaluatee

MethodEndpointหมายเหตุ
PUT/account/passwordส่ง current_password, new_password อย่างน้อย 8 ตัว ผ่านแล้วตั้ง must_change_password เป็น false
GET, PUT/account/profileดูและแก้ full_name, department ของตัวเอง
GET/my/indicatorsตัวชี้วัดของรอบปัจจุบัน พร้อมชื่อหัวข้อ
GET/my/works?indicator_id=ผลงานของตัวเองในตัวชี้วัดนั้น
POST/my/worksindicator_id, title, detail
PUT, DELETE/my/works/:idแก้หรือลบได้เฉพาะของตัวเอง

ทุก SQL ของ /my/works ต้องมี AND evaluatee_id = ? ด้วย req.user.id เสมอ ถ้าเป็นผลงานของคนอื่น ให้ได้ผลเหมือนไม่มีแถวนั้น คือ 404

หน้าเว็บใน pages/evaluatee/: หน้าเปลี่ยนรหัส หน้าข้อมูลส่วนตัว และหน้าผลงานที่เลือกตัวชี้วัดแล้วเพิ่ม แก้ ลบผลงานได้หลายรายการ

ต้องขอคนแกนเพิ่มใน router.js: ถ้า user.must_change_password เป็นจริง ให้ทุกหน้าพาไป /change-password ก่อน และเมื่อเปลี่ยนรหัสสำเร็จ ให้แก้ค่านี้ใน localStorage เป็น false ด้วย

คำใบ้: เปลี่ยนรหัสผ่าน

ดึงแถวของ req.user.id มาก่อน ใช้ bcrypt.compare ตรวจรหัสเดิม ไม่ตรงตอบ 400 ตรงแล้วค่อย bcrypt.hash รหัสใหม่และ UPDATE

ผ่านเมื่อ Login eva1 ครั้งแรกถูกบังคับเปลี่ยนรหัส, Login ใหม่ด้วยรหัสใหม่ได้, เพิ่มผลงานสองรายการในตัวชี้วัดเดียวกันได้ และใช้ id ผลงานของคนอื่นแก้แล้วได้ 404

คนกรรมการ

รายชื่อที่ได้รับมอบหมาย และหน้าดูข้อมูลของผู้รับการประเมิน

Express ใน routes/committee.js ใช้ได้เฉพาะ committee

MethodEndpointหมายเหตุ
GET/committee/assignmentsรายชื่อที่ตัวเองได้รับมอบหมาย พร้อมชื่อผู้รับการประเมิน ชื่อรอบ บทบาทประธานหรือกรรมการร่วม และสถานะ
GET/committee/assignments/:idข้อมูลผู้รับการประเมิน หัวข้อ ตัวชี้วัด ผลงาน และคะแนนประเมินตนเอง เฉพาะรายการที่ committee_id เป็นของตัวเอง

หน้าเว็บใน pages/committee/: หน้ารายชื่อที่กดเข้าไปดูรายละเอียดได้ และหน้ารายละเอียดแบบดูอย่างเดียว เรียงตามหัวข้อ ใต้ตัวชี้วัดแต่ละตัวแสดงผลงานของผู้รับการประเมิน การให้คะแนนเริ่มครั้งที่ 9

คำใบ้: SQL รายชื่อที่ได้รับมอบหมาย
SELECT a.id, a.committee_role, a.status,
       u.full_name AS evaluatee_name, u.department,
       r.name AS round_name
FROM assignments a
JOIN users u ON a.evaluatee_id = u.id
JOIN evaluation_rounds r ON a.round_id = r.id
WHERE a.committee_id = ?

JOIN คือการต่อตารางเข้าด้วยกันตามคอลัมน์ที่ระบุ AS คือตั้งชื่อเล่นให้ตารางหรือคอลัมน์ ลองรันใน MySQL ก่อน โดยใส่ id ของ com1 แทน ?

คำใบ้: กันไม่ให้ดูของคนที่ไม่ได้รับมอบหมาย

ดึงแถวจาก assignments ด้วย WHERE id = ? AND committee_id = ? ถ้าไม่เจอตอบ 404 ทันที เจอแล้วค่อยใช้ evaluatee_id กับ round_id ของแถวนั้นไปดึงข้อมูลต่อ

ผ่านเมื่อ Login com1 เห็น eva1 ในรายชื่อพร้อมบทบาทประธาน, กดเข้าไปเห็นผลงานที่ eva1 เพิ่งเพิ่ม และเรียก assignment id ที่ไม่ใช่ของตัวเองได้ 404
