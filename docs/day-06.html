เรื่องใหม่วันนี้

Middleware คือด่านตรวจ

Request เข้ามาผ่านด่านrequireAuth มี token ถูกต้องไหมผ่านด่านrequireRole บทบาทนี้มีสิทธิ์ไหมผ่านหมดRoute ทำงาน

Middleware คือฟังก์ชันที่ทำงานก่อน Route ถ้าไม่ผ่าน จะตอบกลับไปเลยโดยไม่ให้ถึง Route ถ้าผ่าน จะเรียก next() เพื่อส่งต่อด่านถัดไป

401 กับ 403 ต่างกัน

Codeความหมายตัวอย่าง
401ไม่รู้ว่าเป็นใครไม่แนบ token หรือ token ปลอมหรือหมดอายุ
403รู้ว่าเป็นใคร แต่ไม่มีสิทธิ์กรรมการพยายามเพิ่มตัวชี้วัด

หน้าเว็บส่ง token ยังไง

ทุก request ที่ต้อง Login แนบ header ชื่อ Authorization ค่าเป็น Bearer เว้นวรรคแล้วตามด้วย token เราจะเขียนฟังก์ชัน api() ไว้ที่เดียวให้แนบให้อัตโนมัติ

สร้าง Middleware

ใน backend สร้างโฟลเดอร์ middleware แล้วสร้างไฟล์ middleware/auth.js

const jwt = require('jsonwebtoken');
const { JWT_SECRET } = require('../config');

function requireAuth(req, res, next) {
  const header = req.headers.authorization || '';
  const [type, token] = header.split(' ');
  if (type !== 'Bearer' || !token) {
    return res.status(401).json({ message: 'กรุณาเข้าสู่ระบบ' });
  }
  try {
    req.user = jwt.verify(token, JWT_SECRET);
    next();
  } catch (err) {
    return res.status(401).json({ message: 'token ไม่ถูกต้องหรือหมดอายุ' });
  }
}

function requireRole(...roles) {
  return (req, res, next) => {
    if (!roles.includes(req.user.role)) {
      return res.status(403).json({ message: 'ไม่มีสิทธิ์เข้าถึง' });
    }
    next();
  };
}

module.exports = { requireAuth, requireRole };

สร้าง src/api.js ทุกหน้าเรียก API ผ่านฟังก์ชันนี้ จะได้แนบ token และจัดการ error เหมือนกันทุกที่

const API = import.meta.env.VITE_API_URL

export async function api(path, options = {}) {
  const token = localStorage.getItem('token')
  const res = await fetch(`${API}${path}`, {
    ...options,
    headers: {
      'Content-Type': 'application/json',
      ...(token ? { Authorization: `Bearer ${token}` } : {}),
      ...options.headers
    }
  })
  const data = await res.json().catch(() => ({}))
  if (res.status === 401 && token) {
    localStorage.removeItem('token')
    localStorage.removeItem('user')
    window.location.href = '/login'
  }
  if (!res.ok) throw new Error(data.message || 'เกิดข้อผิดพลาด')
  return data
}

ตั้ง Router และด่านตรวจหน้าเว็บ

สร้าง src/router.js

import { createRouter, createWebHistory } from 'vue-router'
import { homeByRole } from './roles'
import LoginPage from './pages/LoginPage.vue'
import IndicatorsPage from './pages/hr/IndicatorsPage.vue'
import EvaluateeHome from './pages/evaluatee/EvaluateeHome.vue'
import CommitteeHome from './pages/committee/CommitteeHome.vue'

const routes = [
  { path: '/login', component: LoginPage },
  { path: '/hr/indicators', component: IndicatorsPage, meta: { role: 'hr' } },
  { path: '/evaluatee', component: EvaluateeHome, meta: { role: 'evaluatee' } },
  { path: '/committee', component: CommitteeHome, meta: { role: 'committee' } },
  { path: '/:pathMatch(.*)*', redirect: '/login' }
]

const router = createRouter({ history: createWebHistory(), routes })

router.beforeEach((to) => {
  const user = JSON.parse(localStorage.getItem('user') || 'null')
  if (to.path === '/login') return user ? homeByRole[user.role] : true
  if (!user) return '/login'
  if (to.meta.role && to.meta.role !== user.role) return homeByRole[user.role]
  return true
})

export default router

หน้า Login

<script setup>
import { ref } from 'vue'
import { useRouter } from 'vue-router'
import { api } from '../api'
import { homeByRole } from '../roles'

const router = useRouter()
const form = ref({ username: '', password: '' })
const errorMsg = ref('')

async function login() {
  errorMsg.value = ''
  try {
    const data = await api('/auth/login', {
      method: 'POST',
      body: JSON.stringify(form.value)
    })
    localStorage.setItem('token', data.token)
    localStorage.setItem('user', JSON.stringify(data.user))
    router.push(homeByRole[data.user.role])
  } catch (err) {
    errorMsg.value = err.message
  }
}
</script>

<template>
  <main>
    <h1>เข้าสู่ระบบ</h1>
    <form @submit.prevent="login">
      <input v-model="form.username" placeholder="ชื่อผู้ใช้" required>
      <input v-model="form.password" type="password" placeholder="รหัสผ่าน" required>
      <button type="submit">เข้าสู่ระบบ</button>
    </form>
    <p v-if="errorMsg" style="color: red">{{ errorMsg }}</p>
  </main>
</template>

<script setup>
import { ref, onMounted } from 'vue'
import { useRouter } from 'vue-router'
import { api } from '../../api'

const router = useRouter()
const user = JSON.parse(localStorage.getItem('user') || 'null')
const indicators = ref([])
const form = ref({ title: '', weight: 1 })
const editingId = ref(null)
const errorMsg = ref('')

async function load() {
  try {
    indicators.value = await api('/indicators')
  } catch (err) {
    errorMsg.value = err.message
  }
}

async function save() {
  errorMsg.value = ''
  try {
    const path = editingId.value ? `/indicators/${editingId.value}` : '/indicators'
    const method = editingId.value ? 'PUT' : 'POST'
    await api(path, { method, body: JSON.stringify(form.value) })
    cancelEdit()
    await load()
  } catch (err) {
    errorMsg.value = err.message
  }
}

function startEdit(item) {
  editingId.value = item.id
  form.value = { title: item.title, weight: item.weight }
}

function cancelEdit() {
  editingId.value = null
  form.value = { title: '', weight: 1 }
}

async function remove(id) {
  if (!confirm('ยืนยันการลบ?')) return
  try {
    await api(`/indicators/${id}`, { method: 'DELETE' })
    await load()
  } catch (err) {
    errorMsg.value = err.message
  }
}

function logout() {
  localStorage.removeItem('token')
  localStorage.removeItem('user')
  router.push('/login')
}

onMounted(load)
</script>

<template>
  <main>
    <p>สวัสดี {{ user?.full_name }} <button @click="logout">ออกจากระบบ</button></p>
    <h1>ตัวชี้วัด</h1>
    <form @submit.prevent="save">
      <input v-model="form.title" placeholder="ชื่อตัวชี้วัด">
      <input v-model.number="form.weight" type="number" placeholder="น้ำหนัก">
      <button type="submit">{{ editingId ? 'บันทึกการแก้ไข' : 'เพิ่ม' }}</button>
      <button v-if="editingId" type="button" @click="cancelEdit">ยกเลิก</button>
    </form>
    <p v-if="errorMsg" style="color: red">{{ errorMsg }}</p>
    <table border="1" cellpadding="6">
      <thead><tr><th>ลำดับ</th><th>ตัวชี้วัด</th><th>น้ำหนัก</th><th>จัดการ</th></tr></thead>
      <tbody>
        <tr v-for="item in indicators" :key="item.id">
          <td>{{ item.id }}</td>
          <td>{{ item.title }}</td>
          <td>{{ item.weight }}</td>
          <td>
            <button @click="startEdit(item)">แก้ไข</button>
            <button @click="remove(item.id)">ลบ</button>
          </td>
        </tr>
      </tbody>
    </table>
  </main>
</template>
